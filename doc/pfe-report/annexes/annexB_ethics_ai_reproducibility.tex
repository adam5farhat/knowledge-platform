\section{Minimum reproducibility expectations}

\begin{itemize}[leftmargin=*,itemsep=0.2em]
  \item \textbf{Repository:} \texttt{c:\textbackslash dev\textbackslash finalproject} (monorepo);
  \item \textbf{Database:} PostgreSQL 16 + pgvector; migrations via \texttt{npm run db:migrate}; seed via \texttt{npm run db:seed};
  \item \textbf{Services:} \texttt{npm run docker:up} for PostgreSQL and Redis;
  \item \textbf{Environment:} \tcode{apps/api/.env} (\tcode{JWT\_SECRET}, \tcode{GEMINI\_API\_KEY}, \tcode{DATABASE\_URL}); \tcode{apps/web/.env.local} (\tcode{NEXT\_PUBLIC\_API\_URL});
  \item \textbf{Run:} \texttt{npm run dev} from repository root;
  \item \textbf{Tests:} \texttt{npm run test:api}; RAG eval: \texttt{npm run eval:rag:quick} or \texttt{npm run eval:rag};
  \item \textbf{Prompt versioning:} \texttt{RAG\_PROMPT\_VERSION} environment variable invalidates answer cache.
\end{itemize}
